  extends React.ComponentPropsWithoutRef<typeof BaseButton> {

- `error`: `bg-error text-error-on shadow-sm hover:bg-error-hover active:scale-[0.98] active:bg-error-hover`

**Size styles** — define INSIDE the component function (variant-dependent):
```typescript
const sizeStyles: Record<string, string> = {
  sm: variant === "icon" ? "h-8 w-8 p-0" : "h-8 px-3 text-sm",
  md: variant === "icon" ? "h-10 w-10 p-0" : "h-10 px-4 text-base",
  lg: variant === "icon" ? "h-12 w-12 p-0" : "h-12 px-6 text-lg",
}
```

**Component function**:
```typescript
export function Button({
  className,
  classNames,
  variant = "primary",
  size = "md",
  children,
  ...props
}: ButtonProps) {
  const sizeStyles: Record<string, string> = {
    sm: variant === "icon" ? "h-8 w-8 p-0" : "h-8 px-3 text-sm",
    md: variant === "icon" ? "h-10 w-10 p-0" : "h-10 px-4 text-base",
    lg: variant === "icon" ? "h-12 w-12 p-0" : "h-12 px-6 text-lg",
  }

  return (
    <BaseButton
      className={twMerge(
        BASE_STYLES,
        VARIANT_STYLES[variant],
        sizeStyles[size],
        classNames?.root,
        className,
      )}
      {...props}
    >
      {children}
    </BaseButton>
  )
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| className prop (consumer → component) | Consumer passes arbitrary string; React sets it as a DOM attribute |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-02-01 | Injection | className / classNames.root props | accept | React sets className as a DOM attribute value, not innerHTML. CSS injection via className is not possible in React — the value goes to `element.className`, not `element.innerHTML`. No XSS risk. |
| T-01-02-02 | Tampering | children prop (Button) | accept | children is React.ReactNode rendered inside a button element. React escapes all string children. No script injection path. |

Phase 1 plan 02 creates pure presentational components with no data processing, no user input beyond className strings, no network calls. ASVS L1 threat surface is negligible.
</threat_model>
